## Funciones

``` r
histogram_matrix <- function(df, var_num){
  lista_plots <- list()
  for(i in 1:length(var_num)){
    lista_plots[[i]] <- dataset_train |> 
      ggplot(aes(!!sym(var_num[i]))) + 
      geom_histogram(fill = "steelblue") +
      labs(title = var_num[i]) +
      theme(axis.title.x = element_blank())
  }
  
  salida <- patchwork::wrap_plots(lista_plots)
  
  return(salida)
}
```
